# mmCal V1.5 angle semantics

angleMode[]=>Rad
sin[Pi/6]=>1/2
atan[1]=>Pi/4
arg[I]=>Pi/2 Rad
vangle[{1,0},{0,1}]=>Pi/2

angleMode[Deg]=>Deg
angleMode[]=>Deg
sin[30]=>1/2
atan[1]=>45
arg[I]=>Pi/2 Rad
vangle[{1,0},{0,1}]=>90

angleMode[Grad]=>Grad
angleMode[]=>Grad
sin[100]=>1
atan[1]=>50
arg[I]=>Pi/2 Rad
vangle[{1,0},{0,1}]=>100

angleMode[Rad]=>Rad
angleMode[]=>Rad

# Explicit units override the session default.
sin[30 Deg]=>1/2
sin[100 Grad]=>1
