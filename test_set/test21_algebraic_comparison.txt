# AlgebraicNumber Stage 3: exact equality and Real ordering
root[{-2,0,1},2]>1 ==> True
root[{-2,0,1},1]<-1 ==> True
root[{-2,0,1},2]==1 ==> False
root[{-2,0,1},2]!=root[{-3,0,1},2] ==> True
root[{-2,0,1},1]<root[{-2,0,1},2] ==> True
root[{1,0,1},1,Complex]!=root[{1,0,1},2,Complex] ==> True
root[{1,0,1},1,Complex]<root[{1,0,1},2,Complex] ==> root[{1, 0, 1}, 1, Complex] < root[{1, 0, 1}, 2, Complex]
