# Exact symbolic calculus / solver regression.

# Derivatives remain exact.
D[x^5,x]==>5x^4
D[x^5,{x,0}]==>x^5
D[x^5,{x,1}]==>5x^4
D[x^5,{x,5}]==>120
D[x^5,{x,6}]==>0
D[sin[x],{x,4}]==>sin[x]
D[x^2 y^3,x,y]==>6x y^2
D[exp[x^2],x]==>2x exp[x^2]

# Symbolic / exact integration.
integrate[x^2,x]==>x^3/3
integrate[(2x+3)^5,x]==>(3+2x)^6/12
integrate[1/(2x+3),x]==>log[3+2x]/2
integrate[exp[2x+1],x]==>exp[1+2x]/2
integrate[sin[2x],x]==>-cos[2x]/2
integrate[x exp[x],x]==>x exp[x]-exp[x]
integrate[E^x cos[x],x]==>(cos[x]+sin[x])exp[x]/2
integrate[E^x cos[x],x]==(cos[x]+sin[x])/2exp[x]==>True
integrate[x^2,{x,0,1}]==>1/3
integrate[sin[x],{x,0,Pi}]==>2
integrate[1/x,{x,1,2}]==>log[2]
integrate[1/sqrt[x],{x,0,1}]==>2

# Exact limits, including one-sided infinity boundaries.
limit[sin[x]/x,x,0]==>1
limit[(1-cos[x])/x^2,x,0]==>1/2
limit[1/x,x,0,1]==>Infinity
limit[1/x,x,0,-1]==>-Infinity
limit[atan[x],x,Infinity]==>Pi/2
limit[exp[-x],x,Infinity]==>0

# Solve should distinguish empty / finite / conditional domains.
solve[x==1,x]==>{x==1}
solve[x^2==1,x]==>{x==1,x==-1}
solve[x^2+1==0,x,Real]==>{}
solve[x^2+1==0,x,Complex]==>{x==I,x==-I}
solve[2x+3==7,x]==>{x==2}
solve[x^2<4,x]==>{x in Real if x>-2&&x<2}
solve[exp[x]==2,x,Real]==>{x==log[2]}
solve[tanh[x]==1/2,x,Real]==>{x==atanh[1/2]}
solve[tanh[x]==2,x,Real]==>{}
