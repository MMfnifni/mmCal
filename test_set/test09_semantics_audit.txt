integrate[D[x^2,x],x]==>x^2
integrate[normal[series[exp[x],{x,0,2}]],x]==>x+x^2/2+x^3/6
normal[series[D[exp[x],x],{x,0,5}]]==>1+x+x^2/2+x^3/6+x^4/24+x^5/120
normal[series[normal[series[exp[x],{x,0,5}]],{x,0,3}]]==>1+x+x^2/2+x^3/6
solve[D[x^2,x]==2*x,x,Real]==>All
solve[normal[series[exp[x],{x,0,2}]]==1+x+x^2/2,x,Real]==>All

normal[series[1+D[exp[x],x],{x,0,3}]]==>2+x+x^2/2+x^3/6

D[1+normal[series[exp[x],{x,0,2}]],x]==>x+1
D[sin[normal[series[x,{x,0,2}]]],x]==>cos[x]
